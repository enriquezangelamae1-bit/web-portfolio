Java Code – Cash Register Transaction Analyzer

This is my Java console program. It accepts 15 positive transaction amounts, calculates the total and average, counts transactions worth at least ₱1,000, finds the smallest amount, and displays the amounts from highest to lowest. 


        
        import java.util.Scanner;

public class CashRegisterTransaction {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        double[] amount = new double[15];

        double total = 0;
        double average;
        double smallest;
        int count = 0;

        System.out.println("CASH REGISTER TRANSACTION ANALYZER");
        System.out.println("----------------------------------");

        for (int i = 0; i < 15; i++) {

            while (true) {
                System.out.print("Enter transaction " + (i + 1) + ": ");

                if (input.hasNextDouble()) {
                    amount[i] = input.nextDouble();

                    if (amount[i] > 0) {
                        break;
                    } else {
                        System.out.println("Invalid! Enter an amount greater than zero.");
                    }

                } else {
                    System.out.println("Invalid input! Please enter a number.");
                    input.next();
                }
            }

            total += amount[i];

            if (amount[i] >= 1000) {
                count++;
            }
        }

        smallest = amount[0];

        for (int i = 1; i < 15; i++) {
            if (amount[i] < smallest) {
                smallest = amount[i];
            }
        }

        average = total / 15;

        for (int i = 0; i < 14; i++) {
            for (int j = 0; j < 14 - i; j++) {

                if (amount[j] < amount[j + 1]) {

                    double temp = amount[j];
                    amount[j] = amount[j + 1];
                    amount[j + 1] = temp;
                }
            }
        }

        System.out.println("\n========== RESULTS ==========");

        System.out.printf("Total Amount Collected: P%.2f%n", total);
        System.out.printf("Average Transaction: P%.2f%n", average);
        System.out.println("Transactions of P1,000 or more: " + count);
        System.out.printf("Smallest Transaction: P%.2f%n", smallest);

        System.out.println("\nTransactions in Descending Order:");

        for (int i = 0; i < 15; i++) {
            System.out.printf("Index %d: P%.2f%n", i, amount[i]);
        }

        System.out.println("=============================");

        input.close();
    }
}
